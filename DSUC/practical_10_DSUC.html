Aim: Write a program to implement different Sorting Algorithms.

🖥️ Requirements

Personal Computer or Laptop, C compiler (Turbo C/C++ or Visual Studio Code with C extension).

📌 Purpose

To understand and implement the practical concept of sorting algorithms using C language.

Introduction / Theory

Sorting arranges data elements in a particular order, commonly ascending or descending. Common algorithms include Bubble Sort, Selection Sort and Insertion Sort.

C Program

c 
#include <stdio.h>

void bubbleSort(int a[], int n) {
    int i, j, t;
    for (i = 0; i < n - 1; i++)
        for (j = 0; j < n - i - 1; j++)
            if (a[j] > a[j + 1]) {
                t = a[j]; a[j] = a[j + 1]; a[j + 1] = t;
            }
}

void selectionSort(int a[], int n) {
    int i, j, min, t;
    for (i = 0; i < n - 1; i++) {
        min = i;
        for (j = i + 1; j < n; j++)
            if (a[j] < a[min]) min = j;
        t = a[i]; a[i] = a[min]; a[min] = t;
    }
}

void insertionSort(int a[], int n) {
    int i, j, key;
    for (i = 1; i < n; i++) {
        key = a[i];
        j = i - 1;
        while (j >= 0 && a[j] > key) {
            a[j + 1] = a[j];
            j--;
        }
        a[j + 1] = key;
    }
}

int main() {
    int a[100], n, i;
    printf("Enter number of elements: ");
    scanf("%d", &n);
    for (i = 0; i < n; i++) scanf("%d", &a[i]);

    bubbleSort(a, n);

    printf("Sorted array: ");
    for (i = 0; i < n; i++) printf("%d ", a[i]);
    return 0;
}
