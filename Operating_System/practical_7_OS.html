🖥️ Requirements

📌 Purpose

To understand and implement contiguous memory allocation techniques and observe how memory blocks are assigned to processes using First Fit, Best Fit and Worst Fit methods.

Introduction

Contiguous memory allocation is a memory management technique in which each process is allocated a continuous block of memory. The operating system uses different allocation strategies to assign available memory blocks to processes.


The three commonly used contiguous memory allocation techniques are First Fit, Best Fit and Worst Fit. First Fit allocates the first suitable memory block, Best Fit allocates the smallest suitable block, and Worst Fit allocates the largest suitable block.

Memory Allocation Techniques

01

First Fit

First Fit searches the memory blocks from the beginning and allocates the first block that is large enough to accommodate the process.

02

Best Fit

Best Fit searches all available memory blocks and allocates the smallest block that is large enough for the process.

03

Worst Fit

Worst Fit searches the available memory blocks and allocates the largest block that is large enough for the process.

#include <stdio.h>
 

void firstFit(int blockSize[], int m, int processSize[], int n) {

 
int allocation[n];

for (int i = 0; i < n; i++)
    allocation[i] = -1;

printf("\n================ FIRST FIT STEP TABLE ================\n");
printf("-------------------------------------------------------------\n");
printf("| Process | Process Size | Block Allocated | Remaining Block Size |\n");
printf("-------------------------------------------------------------\n");

for (int i = 0; i < n; i++) {

    int allocated = 0;

    for (int j = 0; j < m; j++) {

        if (blockSize[j] >= processSize[i]) {

            allocation[i] = j;

            printf("|   P%-5d | %-12d | Block %-9d | %d - %d = %d |\n",
                   i + 1,
                   processSize[i],
                   j + 1,
                   blockSize[j],
                   processSize[i],
                   blockSize[j] - processSize[i]);

            blockSize[j] -= processSize[i];

            allocated = 1;
            break;
        }
    }

    if (!allocated) {

        printf("|   P%-5d | %-12d | Not Allocated   | --- |\n",
               i + 1,
               processSize[i]);
    }
}

printf("-------------------------------------------------------------\n");

printf("\nFinal First Fit Allocation:\n");
printf("Process No.\tProcess Size\tBlock No.\n");

for (int i = 0; i < n; i++) {

    printf("%d\t\t%d\t\t",
           i + 1,
           processSize[i]);

    if (allocation[i] != -1)
        printf("%d\n", allocation[i] + 1);
    else
        printf("Not Allocated\n");
}
 

}

void bestFit(int blockSize[], int m, int processSize[], int n) {

 
int allocation[n];

for (int i = 0; i < n; i++)
    allocation[i] = -1;

printf("\n================ BEST FIT STEP TABLE ================\n");
printf("---------------------------------------------------------------------\n");
printf("| Process | Process Size | Block Allocated (Best) | Remaining Block Size |\n");
printf("---------------------------------------------------------------------\n");

for (int i = 0; i < n; i++) {

    int bestIdx = -1;

    for (int j = 0; j < m; j++) {

        if (blockSize[j] >= processSize[i]) {

            if (bestIdx == -1 ||
                blockSize[j] < blockSize[bestIdx]) {

                bestIdx = j;
            }
        }
    }

    if (bestIdx != -1) {

        printf("|   P%-5d | %-12d | Block %-18d | %d - %d = %d |\n",
               i + 1,
               processSize[i],
               bestIdx + 1,
               blockSize[bestIdx],
               processSize[i],
               blockSize[bestIdx] - processSize[i]);

        allocation[i] = bestIdx;

        blockSize[bestIdx] -= processSize[i];

    } else {

        printf("|   P%-5d | %-12d | Not Allocated        | --- |\n",
               i + 1,
               processSize[i]);
    }
}

printf("---------------------------------------------------------------------\n");

printf("\nFinal Best Fit Allocation:\n");
printf("Process No.\tProcess Size\tBlock No.\n");

for (int i = 0; i < n; i++) {

    printf("%d\t\t%d\t\t",
           i + 1,
           processSize[i]);

    if (allocation[i] != -1)
        printf("%d\n", allocation[i] + 1);
    else
        printf("Not Allocated\n");
}
 

}
