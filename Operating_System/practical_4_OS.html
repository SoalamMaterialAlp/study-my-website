Aim: Simulate the following CPU scheduling algorithms: FCFS, SJF, Round Robin and Priority Scheduling.

🖥️ Requirements

Personal Computer or Laptop and C Compiler (Turbo C++ or Visual Studio Code with C extension).

📌 Purpose

To implement and simulate different CPU scheduling algorithms and calculate Completion Time, Turnaround Time, Waiting Time and their average values.

Introduction

CPU scheduling is the process of selecting a process from the ready queue and allocating the CPU to it. Different scheduling algorithms use different methods to decide which process should execute next.


In this practical, four CPU scheduling algorithms are simulated: First Come First Serve (FCFS), Shortest Job First (SJF), Round Robin and Priority Scheduling. The programs calculate Completion Time (CT), Turnaround Time (TAT), Waiting Time (WT), and average TAT and WT.

Procedure

01

FCFS Algorithm

First Come First Serve (FCFS) scheduling executes processes in the order of their arrival time. Enter the number of processes, arrival time and burst time, then calculate CT, TAT and WT.

c 
#include <stdio.h>

int main() {

    int n, i, j;
    printf("FCFS Alghorithm scheduling By Kadir Alam\n");

    printf("Enter the number of processes: ");
    scanf("%d", &n);

    int process[n], arrival_time[n], burst_time[n];
    int completion_time[n], waiting_time[n];
    int turnaround_time[n];
    int temp;

    // Taking input
    for(i = 0; i < n; i++) {

        process[i] = i + 1;

        printf("\nEnter Arrival Time of P%d: ", i + 1);
        scanf("%d", &arrival_time[i]);

        printf("Enter Burst Time of P%d: ", i + 1);
        scanf("%d", &burst_time[i]);
    }

    // Sorting by Arrival Time
    for(i = 0; i < n; i++) {

        for(j = i + 1; j < n; j++) {

            if(arrival_time[i] > arrival_time[j]) {

                temp = arrival_time[i];
                arrival_time[i] = arrival_time[j];
                arrival_time[j] = temp;

                temp = burst_time[i];
                burst_time[i] = burst_time[j];
                burst_time[j] = temp;

                temp = process[i];
                process[i] = process[j];
                process[j] = temp;
            }
        }
    }

    // Calculate Completion Time
    completion_time[0] =
        arrival_time[0] + burst_time[0];

    for(i = 1; i < n; i++) {

        if(completion_time[i-1] < arrival_time[i]) {

            completion_time[i] =
                arrival_time[i] + burst_time[i];

        } else {

            completion_time[i] =
                completion_time[i-1] + burst_time[i];
        }
    }

    // Calculate TAT and WT
    for(i = 0; i < n; i++) {

        turnaround_time[i] =
            completion_time[i] - arrival_time[i];

        waiting_time[i] =
            turnaround_time[i] - burst_time[i];
    }

    // Display table
    printf("\nProcess\tAT\tBT\tCT\tTAT\tWT\n");

    for(i = 0; i < n; i++) {

        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
               process[i],
               arrival_time[i],
               burst_time[i],
               completion_time[i],
               turnaround_time[i],
               waiting_time[i]);
    }

    // Average calculation
    float avg_wt = 0, avg_tat = 0;

    for(i = 0; i < n; i++) {

        avg_wt += waiting_time[i];
        avg_tat += turnaround_time[i];
    }

    avg_wt /= n;
    avg_tat /= n;

    printf("\nAverage Waiting Time: %.2f", avg_wt);
    printf("\nAverage Turnaround Time: %.2f\n", avg_tat);

    return 0;
}
02

SJF Algorithm

Shortest Job First (SJF) scheduling selects the process having the shortest burst time among the processes that have already arrived.

c 
#include <stdio.h>

int main() {

    int n, i, j;

    float avg_tat = 0, avg_wt = 0;

    printf("Shortest Job First Algorithm Scheduling by Kadir Alam\n");

    printf("Enter number of processes: ");
    scanf("%d", &n);

    int pid[n], at[n], bt[n], ct[n];
    int tat[n], wt[n], completed[n];

    for(i = 0; i < n; i++) {

        pid[i] = i + 1;

        printf("\nEnter Arrival Time of P%d: ", i + 1);
        scanf("%d", &at[i]);

        printf("Enter Burst Time of P%d: ", i + 1);
        scanf("%d", &bt[i]);

        completed[i] = 0;
    }

    int current_time = 0;
    int completed_count = 0;

    while(completed_count < n) {

        int idx = -1;
        int min_bt = 1000000000;

        // Find process with minimum BT
        for(i = 0; i < n; i++) {

            if(at[i] <= current_time &&
               completed[i] == 0) {

                if(bt[i] < min_bt) {

                    min_bt = bt[i];
                    idx = i;
                }
            }
        }

        if(idx != -1) {

            current_time += bt[idx];

            ct[idx] = current_time;

            tat[idx] =
                ct[idx] - at[idx];

            wt[idx] =
                tat[idx] - bt[idx];

            completed[idx] = 1;

            completed_count++;

            avg_tat += tat[idx];
            avg_wt += wt[idx];

        } else {

            current_time++;
        }
    }

    printf("\nPID\tAT\tBT\tCT\tTAT\tWT\n");

    for(i = 0; i < n; i++) {

        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
               pid[i],
               at[i],
               bt[i],
               ct[i],
               tat[i],
               wt[i]);
    }

    printf("\nAverage TAT = %.2f",
           avg_tat / n);

    printf("\nAverage WT = %.2f\n",
           avg_wt / n);

    return 0;
}
03

Round Robin Algorithm

Round Robin scheduling gives each process a fixed amount of CPU time called the Time Quantum. If a process is not completed within the time quantum, it gets another turn later.

c 
#include <stdio.h>

int main() {

    int n, tq;

    printf("Rounnd Robin by Kadir Alam\n");

    printf("Enter number of processes: ");
    scanf("%d", &n);

    printf("Enter Time Quantum: ");
    scanf("%d", &tq);

    int pid[n], at[n], bt[n], rt[n];
    int ct[n], tat[n], wt[n];

    float avg_tat = 0, avg_wt = 0;

    for (int i = 0; i < n; i++) {

        pid[i] = i + 1;

        printf("\nEnter Arrival Time of P%d: ", i + 1);
        scanf("%d", &at[i]);

        printf("Enter Burst Time of P%d: ", i + 1);
        scanf("%d", &bt[i]);

        rt[i] = bt[i];

        ct[i] = 0;
    }

    int current_time = 0;
    int completed = 0;
    int done = 0;

    while(completed < n) {

        done = 1;

        for(int i = 0; i < n; i++) {

            if(at[i] <= current_time &&
               rt[i] > 0) {

                done = 0;

                if(rt[i] > tq) {

                    current_time += tq;
                    rt[i] -= tq;

                } else {

                    current_time += rt[i];

                    rt[i] = 0;

                    ct[i] = current_time;

                    tat[i] =
                        ct[i] - at[i];

                    wt[i] =
                        tat[i] - bt[i];

                    avg_tat += tat[i];
                    avg_wt += wt[i];

                    completed++;
                }
            }
        }

        if(done == 1) {

            current_time++;
        }
    }

    printf("\nPID\tAT\tBT\tCT\tTAT\tWT\n");

    for(int i = 0; i < n; i++) {

        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
               pid[i],
               at[i],
               bt[i],
               ct[i],
               tat[i],
               wt[i]);
    }

    printf("\nAverage TAT = %.2f",
           avg_tat / n);

    printf("\nAverage WT = %.2f\n",
           avg_wt / n);

    return 0;
}

Priority Scheduling (Non-Preemptive)

In non-preemptive Priority Scheduling, the process with the highest priority among the arrived processes is selected and executed until completion. In this program, a smaller priority number represents a higher priority.

        scanf("%d", &bt[i]);

        printf("Enter Priority of P%d: ", i + 1);
        scanf("%d", &pr[i]);

        completed[i] = 0;
    }

    int current_time = 0;
    int completed_count = 0;

    while(completed_count < n) {

            if(at[i] <= current_time &&

Priority Scheduling (Preemptive)

In Preemptive Priority Scheduling, the CPU can be taken from the currently running process when a higher-priority process becomes available. In this program, a smaller priority number represents a higher priority.

c 
#include <stdio.h>

int main() {

    int n;

    printf("Priority Scheduling (Preemptive Scheduling) by Kadir Alam\n");

    printf("Enter number of processes: ");
    scanf("%d", &n);

    int pid[n], at[n], bt[n], rt[n], pr[n];
    int ct[n], tat[n], wt[n], completed[n];

    float avg_tat = 0, avg_wt = 0;

    for(int i = 0; i < n; i++) {

        pid[i] = i + 1;

        printf("\nEnter Arrival Time of P%d: ", i + 1);
        scanf("%d", &at[i]);

        printf("Enter Burst Time of P%d: ", i + 1);
        scanf("%d", &bt[i]);

        printf("Enter Priority of P%d: ", i + 1);
        scanf("%d", &pr[i]);

        rt[i] = bt[i];

        completed[i] = 0;
    }

    int current_time = 0;
    int completed_count = 0;

    while(completed_count < n) {

        int idx = -1;
        int best_pr = 1000000000;

        for(int i = 0; i < n; i++) {

            if(at[i] <= current_time &&
               completed[i] == 0) {

                if(pr[i] < best_pr) {

                    best_pr = pr[i];
                    idx = i;
                }
            }
        }

        if(idx != -1) {

            rt[idx]--;

            current_time++;

            if(rt[idx] == 0) {

                completed[idx] = 1;

                completed_count++;

                ct[idx] = current_time;

                tat[idx] =
                    ct[idx] - at[idx];

                wt[idx] =
                    tat[idx] - bt[idx];

                avg_tat += tat[idx];
                avg_wt += wt[idx];
            }

        } else {

            current_time++;
        }
    }

    printf("\nPID\tAT\tBT\tPR\tCT\tTAT\tWT\n");

    for(int i = 0; i < n; i++) {

        printf("P%d\t%d\t%d\t%d\t%d\t%d\t%d\n",
               pid[i],
               at[i],
               bt[i],
               pr[i],
               ct[i],
               tat[i],
               wt[i]);
    }

    printf("\nAverage TAT = %.2f",
           avg_tat / n);

    printf("\nAverage WT = %.2f\n",
           avg_wt / n);

    return 0;
}
