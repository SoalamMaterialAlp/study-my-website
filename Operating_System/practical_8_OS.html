Aim: Simulate all page replacement algorithms: (a) FIFO, (b) LRU, (c) OPTIMAL.

🖥️ Requirements

Personal Computer or Laptop, C Compiler (Turbo C++ or Visual Studio Code with C extension).

📌 Purpose

To understand and simulate different page replacement algorithms used by an Operating System for managing pages in main memory.

Introduction

Page replacement is a memory management technique used by an Operating System when a page fault occurs and all available frames are occupied.


In this practical, three page replacement algorithms are simulated: FIFO (First In First Out), LRU (Least Recently Used) and OPTIMAL page replacement algorithm.


Page Replacement Algorithms

  • FIFO: The page that entered the memory first is replaced first.
  • LRU: The page that has not been used for the longest period of time is replaced.
  • OPTIMAL: The page that will not be used for the longest period in the future is replaced.

(a) FIFO Page Replacement Algorithm

FIFO stands for First In First Out. In this algorithm, the page which entered the memory first is replaced first when a page fault occurs and all frames are occupied.


Algorithm:

  1. Initialize all frames as empty.
  2. Read the page reference string.
  3. Check whether the requested page is already present.
  4. If the page is present, no replacement is performed.
  5. If the page is not present, replace the oldest page.
  6. Increase the page fault count.
  7. Display the current frame status.
c 
#include <stdio.h>

int main()
{
    int frames, pages, page[50], frame[10];
    int i, j, pointer = 0, found, faults = 0;

    printf("FIFO Page Replacement Algorithm Performed By Kadir Alam\n");

    printf("Enter number of frames: ");
    scanf("%d", &frames);

    printf("Enter number of pages: ");
    scanf("%d", &pages);

    printf("Enter page reference string: ");

    for(i = 0; i < pages; i++)
        scanf("%d", &page[i]);

    // Initialize all frames as empty
    for(i = 0; i < frames; i++)
        frame[i] = -1;

    printf("\nStep-by-Step Allocation / Replacement:\n\n");

    for(i = 0; i < pages; i++)
    {
        found = 0;

        // Check if page already exists
        for(j = 0; j < frames; j++)
        {
            if(frame[j] == page[i])
            {
                found = 1;

                printf("Page %d already in memory -> No Replacement\n",
                       page[i]);

                break;
            }
        }

        if(found == 0)
        {
            if(frame[pointer] == -1)
            {
                printf("Page %d allocated to frame %d\n",
                       page[i], pointer);
            }
            else
            {
                printf("Page %d comes -> Replace Page %d with %d\n",
                       page[i], frame[pointer], page[i]);
            }

            frame[pointer] = page[i];

            pointer = (pointer + 1) % frames;

            faults++;
        }

        // Print current frame status
        printf("Frames: ");

        for(j = 0; j < frames; j++)
        {
            if(frame[j] != -1)
                printf("%d ", frame[j]);
            else
                printf("- ");
        }

        printf("\n\n");
    }

    printf("Total Page Faults = %d\n", faults);

    return 0;
}

(b) LRU Page Replacement Algorithm

LRU stands for Least Recently Used. In this algorithm, the page which has not been used for the longest period of time is replaced when a page fault occurs.


Algorithm:

  1. Initialize all frames as empty.
  2. Read the page reference string.
  3. Check whether the requested page is already in memory.
  4. If the page is present, update its recent-use time.
  5. If the page is not present and an empty frame exists, load the page.
  6. If all frames are full, find the least recently used page.
  7. Replace the least recently used page.
  8. Increase the page fault count.
c 
#include <stdio.h>

int findLRU(int used[], int frames)
{
    int i, min = used[0], pos = 0;

    for(i = 1; i < frames; i++)
    {
        if(used[i] < min)
        {
            min = used[i];
            pos = i;
        }
    }

    return pos;
}

int main()
{
    int frames, pages, page[50], frame[10], used[10];
    int i, j, k, flag, pos;
    int faults = 0, time = 0;

    printf("=== LRU Page Replacement Algorithm ===\n");

    printf("LRU Page Replacement Algorithm Performed By Kadir Alam\n");

    printf("Enter number of frames: ");
    scanf("%d", &frames);

    printf("Enter number of pages: ");
    scanf("%d", &pages);

    printf("Enter page reference string: ");

    for(i = 0; i < pages; i++)
        scanf("%d", &page[i]);

    for(i = 0; i < frames; i++)
        frame[i] = -1;

    printf("\nStep-by-Step Allocation & Replacement:\n\n");

    for(i = 0; i < pages; i++)
    {
        flag = 0;

        for(j = 0; j < frames; j++)
        {
            if(frame[j] == page[i])
            {
                used[j] = ++time;
                flag = 1;
                break;
            }
        }

        if(flag == 0)
        {
            for(j = 0; j < frames; j++)
            {
                if(frame[j] == -1)
                {
                    frame[j] = page[i];
                    used[j] = ++time;
                    faults++;

                    printf("Page %d loaded\n", page[i]);

                    goto print_frame;
                }
            }

            pos = findLRU(used, frames);

            printf("Page %d comes -> Replace %d (Least Recently Used)\n",
                   page[i], frame[pos]);

            frame[pos] = page[i];
            used[pos] = ++time;

            faults++;
        }

        print_frame:

        printf("Frames: ");

        for(k = 0; k < frames; k++)
        {
            if(frame[k] == -1)
                printf("- ");
            else
                printf("%d ", frame[k]);
        }

        printf("\nRecent Use Order (MRU -> LRU): ");

        for(k = time; k > 0; k--)
        {
            for(j = 0; j < frames; j++)
            {
                if(used[j] == k)
                    printf(" %d ", frame[j]);
            }
        }

        printf("\n\n");
    }

    printf("Total Page Faults = %d\n", faults);

    return 0;
}

(c) OPTIMAL Page Replacement Algorithm

OPTIMAL page replacement algorithm replaces the page which will not be used for the longest period of time in the future. It uses future page references to decide which page should be replaced.


Algorithm:

  1. Initialize all frames as empty.
  2. Read the page reference string.
  3. Check whether the requested page is already present.
  4. If the page is present, no replacement is performed.
  5. If an empty frame exists, load the requested page.
  6. If all frames are full, check the future references.
  7. Find the page whose next use is farthest in the future.
  8. Replace that page.
  9. Increase the page fault count.
c 
#include <stdio.h>

int findOptimal(int frame[], int page[], int index,
                int frames, int pages)
{
    int i, j, pos = -1, farthest = index;

    for(i = 0; i < frames; i++)
    {
        int found = 0;

        for(j = index + 1; j < pages; j++)
        {
            if(frame[i] == page[j])
            {
                if(j > farthest)
                {
                    farthest = j;
                    pos = i;
                }

                found = 1;
                break;
            }
        }

        if(found == 0)
            return i;
    }

    if(pos == -1)
        return 0;
    else
        return pos;
}
